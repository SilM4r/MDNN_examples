using Microsoft.VisualBasic.FileIO;
using My_DNN.Activation_functions;
using My_DNN.Layers;
using My_DNN.Layers.classes;
using My_DNN.Loss_functions;
using My_DNN.Optimizers;


namespace My_DNN
{
    internal class Program
    {

        static void Main(string[] args)
        {

            double[][][] dataX;
            double[][][] dataY;


            Random rng = new Random();

            Console.WriteLine("loading dataset from disk to ram...");

            double[][][] data = {
                new double[][] { new double[] {0},new double[] {1},new double[] {1},new double[] {2},new double[] {3} },
                new double[][] { new double[] {5},new double[] {8},new double[] {13},new double[] {21},new double[] {34}},
                new double[][] { new double[] {55},new double[] {89},new double[] {144},new double[] {233},new double[] {377}},
                new double[][] { new double[] {610},new double[] {987},new double[] {1597},new double[] {2584},new double[] {4181}},
                new double[][] { new double[] {2},new double[] {3},new double[] {5},new double[] {8},new double[] {13}},
                new double[][] { new double[] {21},new double[] {34},new double[] {55},new double[] {89},new double[] {144}},
                new double[][] { new double[] {233},new double[] {377},new double[] {610},new double[] {987},new double[] {1597}},
            };

            int rows = data.Count();
            int cols = data[0].Count() - 1;

            dataX = new double[rows][][];
            dataY = new double[rows][][];

            for (int i = 0; i < rows; i++)
            {
                dataX[i] = new double[cols][];
                dataY[i] = new double[cols][];
                for (int j = 0; j < cols; j++)
                {
                    dataX[i][j] = new double[data[i][j].Count()];
                    dataY[i][j] = new double[data[i][j].Count()];
                }
            }


            for (int i = 0; i < data.Count(); i++)
            {
                for (int j = 0; j < (data[i].Count() - 1); j++)
                {
                    dataX[i][j] = data[i][j];
                    dataY[i][j] = data[i][j + 1];
                }
            }



            //MDNN model = MDNN.LoadModel("AutoSave.json");


            //model.Context.CalculationViaGpu = true;

            // NORMALIZACE — bez ní se tenhle příklad nenaučí vůbec nic.
            // Fibonacciho hodnoty jdou do 4181, takže MSE počítá s chybami v řádu tisíců
            // a gradienty utečou. Původní verze to řešila hrubou silou (2 260 000 epoch)
            // a stejně skončila s loss ~15 000. Po vydělení maximem (hodnoty do <0,1>)
            // spadne loss na ~0,0006 za pár set epoch.
            double maxValue = 0;
            foreach (double[][] sequence in data)
            {
                foreach (double[] step in sequence)
                {
                    foreach (double value in step)
                    {
                        maxValue = Math.Max(maxValue, value);
                    }
                }
            }

            for (int i = 0; i < rows; i++)
            {
                for (int j = 0; j < cols; j++)
                {
                    for (int k = 0; k < dataX[i][j].Length; k++)
                    {
                        dataX[i][j][k] /= maxValue;
                        dataY[i][j][k] /= maxValue;
                    }
                }
            }

            Console.WriteLine("Convert data to dataset and splitting to input and target output data... done");
            Console.WriteLine("Creating model...");

            Layer outputLayer = new RNN(1, new ReLu());

            // seed = reprodukovatelnost
            // learning rate 0.005 (ne 0.00001) — po normalizaci dat je 1e-5 zbytečně opatrné
            MDNN model = new MDNN(outputLayer, new Adam(0.005), new MSE(), seed: 42);

            model.info();

            model.Train.NumberOfSkipFristEpochInPlotter = 0;
            model.Train.NumberOfShowEpochInConsole = 100;

            Console.WriteLine("Creating model... done");
            Console.WriteLine("Train... ");

            //model.Train.TestNeuralNetwork(dataX.ToArray(), dataY.ToArray());
            // Dvě opravy proti původní verzi:
            //  1) TrainLoop bere pole, ne Tensor — Tensor přetížení neexistuje.
            //  2) isSequence: true. Původně tu bylo false, nejspíš proto, že sekvenční
            //     režim tehdy ještě pořádně nefungoval. Data jsou [7 sekvencí][4 kroky][1],
            //     RNN se má učit krok po kroku a nést si skrytý stav — s false by se každá
            //     sekvence tvářila jako jeden plochý vzorek o 4 hodnotách.
            //
            // POZOR na význam epoch: epocha = jeden PLNÝ průchod trénovacím setem, ne jeden
            // krok optimizeru. Ze 7 sekvencí jdou 4 na trénink, takže při dávce 1 jsou to
            // 4 kroky na epochu. Původních 2 260 000 "epoch" by dnes byla víc než půlmilionová
            // hodnota. Po normalizaci dat stačí 2000 epoch (= 8000 kroků), loss klesne
            // na ~0,0006 a síť předpovídá další člen posloupnosti správně.
            model.Train.TrainLoop(dataX, dataY, 2000, 1, true);

            Console.WriteLine("Train... done ");

            //model.SaveAsJson("MNIST2");

        }

    }
}
