using My_DNN.Activation_functions;
using My_DNN.Layers;
using My_DNN.Loss_functions;
using My_DNN.Optimizers;
using System;


namespace My_DNN
{
    internal class Program
    {
        static void Main(string[] args)
        {
            int totalEpoch = 90000;
            int size_of_mini_batch = 5;
            int maxValue = 10;

            Random rnd = new Random(42);

            //model.Context.CalculationViaGpu = true;

            // seed = reprodukovatelnost
            MDNN model = new MDNN(new Dense(1, new Linear()), new Adam(0.001), new MSE(), seed: 42);
            model.Layers.Add(new Dense(32, new Leak_ReLu()));
            model.Layers.Add(new Dense(32, new Leak_ReLu()));
            model.Layers.Add(new Dense(32, new Leak_ReLu()));
            model.Layers.Add(new Dense(32, new Leak_ReLu()));

            model.Train.TotalEpoch = (uint)totalEpoch;
            model.Train.MiniBatch = (uint)size_of_mini_batch;


            model.info();

            for (int epoch = 0; epoch < totalEpoch; epoch++)
            {
                for (uint miniBatch = 0; miniBatch < size_of_mini_batch; miniBatch++)
                {
                    int num1 = rnd.Next(maxValue);
                    int num2 = rnd.Next(maxValue);

                    model.Train.Fit(new Tensor(new double[] { num1, num2 }), new Tensor (new double[] { num1 * num2 }));
                }

                model.Train.UpdateParams();

                if (epoch % 1000 == 0)
                {
                    // loss se bere z modelu (model.Loss), ne z globálního nastavení —
                    // od zavedení NetworkContextu má každý model tu svou.
                    double loss = model.Loss.GetAverageLossPerIteration();
                    if (!double.IsFinite(loss))
                    {
                        Console.WriteLine("Error: Nan number or infinity number");
                        return;
                    }

                    // Ruční smyčka žádné epochy nemá, takže CurrentEpoch zůstane 0 —
                    // smysl dává počítadlo kroků optimizeru.
                    Console.WriteLine($"krok {model.Train.OptimizerSteps,7} / {totalEpoch}   loss {loss:F5}");
                }
            }

            Console.WriteLine(control(model));
            Console.WriteLine("Total variable = " + maxValue * maxValue);

            model.SaveAsJson("computinTest");

            // Interaktivní vyzkoušení. Přeskočí se, když vstup není terminál (např. při
            // spuštění ze skriptu) — jinak by se program zasekl na čekání na vstup.
            while (!Console.IsInputRedirected)
            {
                Console.Write("první číslo (prázdné = konec): ");
                string? first = Console.ReadLine();
                if (string.IsNullOrWhiteSpace(first)) break;

                Console.Write("druhé číslo: ");
                string? second = Console.ReadLine();
                if (string.IsNullOrWhiteSpace(second)) break;

                int num1 = int.Parse(first);
                int num2 = int.Parse(second);
                Console.WriteLine(model.GetResults(new Tensor(new double[] { num1, num2 })).Data[0]);
            }

            int control(MDNN model)
            {
                int score = 0;

                for (int i = 0; i < maxValue; i++)
                {
                    for (int j = 0; j < maxValue; j++)
                    {
                        double[] z = model.GetResults(new Tensor(new double[] { i, j })).Data;

                        if (Math.Round(z[0]) == i * j)
                        {
                            score++;
                        }
                    }

                }

                return score;
            }
        }

    }
}
